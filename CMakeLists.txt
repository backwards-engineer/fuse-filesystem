cmake_minimum_required(VERSION 3.22)
project(local_fusefs VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BUILD_FUSE "Build the Linux FUSE executable" ON)
add_library(fs_core src/fs_core.c)
target_include_directories(fs_core PUBLIC src)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fs_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE fs_core)
  add_test(NAME filesystem_core COMMAND core_test)
endif()
if(BUILD_FUSE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "FUSE requires Linux. Use WSL2, or -DBUILD_FUSE=OFF for portable core tests.")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3>=3.10)
  add_executable(local-fusefs src/main.c)
  target_compile_definitions(local-fusefs PRIVATE _FILE_OFFSET_BITS=64 _DEFAULT_SOURCE)
  target_compile_options(local-fusefs PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(local-fusefs PRIVATE fs_core PkgConfig::FUSE3)
  install(TARGETS local-fusefs RUNTIME DESTINATION bin)
  install(FILES README.md DESTINATION share/doc/local-fusefs)
  set(CPACK_GENERATOR "TGZ;DEB")
  set(CPACK_PACKAGE_CONTACT "Local FUSE project <developer@localhost>")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "fuse3")
  set(CPACK_PACKAGE_FILE_NAME "local-fusefs-${PROJECT_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
  include(CPack)
endif()

cmake_minimum_required(VERSION 3.22)
project(local_fusefs VERSION 0.2.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build local-fusefs on Linux or inside WSL2.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED)
find_package(SQLite3 3.37 REQUIRED)
pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3>=3.12)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_compile_definitions(_FILE_OFFSET_BITS=64 _GNU_SOURCE)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(fs_core src/store.cpp src/crypto.cpp)
target_include_directories(fs_core PUBLIC src)
target_link_libraries(fs_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto PkgConfig::ZSTD Threads::Threads)
add_executable(local-fusefs src/main.cpp)
target_link_libraries(local-fusefs PRIVATE fs_core PkgConfig::FUSE3)
add_executable(fusefs-admin src/admin.cpp)
target_link_libraries(fusefs-admin PRIVATE fs_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE fs_core)
  set(TEST_PREFIX "")
  if(ENABLE_SANITIZERS)
    set(TEST_PREFIX "sanitized_")
  endif()
  set(CORE_CASES namespace chunk_sizes random_io_model sparse_and_holes dedup_compression_cache
    history_gc_restore permissions_symlinks_times rename concurrency exclusive_and_key_checks
    aead corruption extreme_offsets rollback crash_write crash_rename crash_maintenance)
  set(CASE_INDEX 0)
  foreach(CASE IN LISTS CORE_CASES)
    add_test(NAME ${TEST_PREFIX}storage_${CASE} COMMAND core_test ${CASE_INDEX})
    set_tests_properties(${TEST_PREFIX}storage_${CASE} PROPERTIES TIMEOUT 90)
    math(EXPR CASE_INDEX "${CASE_INDEX} + 1")
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(FUSE_CASES streaming_offsets_and_fsync sparse_seek_and_punch truncate_clears_hidden_tail
    permissions_and_ownership umask_and_timestamps symlinks_and_dangling_links
    atomic_rename_and_open_destination rename_flags_and_directory_cycles rename_observers_see_complete_files
    open_unlink_lifetime multithreaded_random_access atomic_append posix_byte_range_locks
    flock_blocking_and_release open_file_description_locks remount_persistence sigkill_after_acknowledged_write
    cli_history_restore_prune_gc cli_import_export_stream corruption_fails_closed
    exclusive_admin_and_wrong_key readonly_mount directory_unicode_and_errors)
  foreach(CASE IN LISTS FUSE_CASES)
    add_test(NAME ${TEST_PREFIX}fuse_${CASE} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:local-fusefs> $<TARGET_FILE:fusefs-admin> FuseTests.test_${CASE})
    set_tests_properties(${TEST_PREFIX}fuse_${CASE} PROPERTIES TIMEOUT 90)
  endforeach()
endif()
add_executable(fusefs-bench benchmarks/bench.cpp)
target_link_libraries(fusefs-bench PRIVATE fs_core)
install(TARGETS local-fusefs fusefs-admin fusefs-bench RUNTIME DESTINATION bin)
install(FILES README.md docs/storage.md docs/benchmarks.md DESTINATION share/doc/local-fusefs)
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_PACKAGE_NAME "local-fusefs")
set(CPACK_PACKAGE_CONTACT "Local FUSE project <developer@localhost>")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "fuse3")
set(CPACK_PACKAGE_FILE_NAME "local-fusefs-${PROJECT_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
